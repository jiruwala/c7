var fnSelRep = function (e) {
  var cont = sap.ui.getCore().byId(e.currentTarget.id
  );
  if (cont.getCustomData()[0].getKey() == UtilGen.DBView.db2_rep)
    return;
  UtilGen.DBView.db2_rep = cont.getCustomData()[0].getKey();
  showRep();
};
var showRep = function () {
  var hb1 = UtilGen.DBView.byId("db2hbReps1");
  if (hb1 != undefined)
    for (var h = hb1.getItems().length - 1; h >= 0; h--)
      hb1.getItems()[h].destroy();

  Util.destroyID("db2hbReps0", UtilGen.DBView);
  Util.destroyID("db2hbReps", UtilGen.DBView);
  Util.destroyID("db2hbReps1", UtilGen.DBView);
  Util.destroyID("db2selCust", UtilGen.DBView);
  Util.destroyID("db2selSupp", UtilGen.DBView);
  Util.destroyID("db2FromTxt", UtilGen.DBView);
  Util.destroyID("db2ToTxt", UtilGen.DBView);
  Util.destroyID("db2RfrTxt", UtilGen.DBView);
  UtilGen.clearPage(sc);

  var t1 = new sap.m.Text(UtilGen.DBView.createId("db2selCust"), { customData: [{ key: 'customers' }], width: "140px", text: Util.getLangText("db2CustReports") }).addStyleClass(UtilGen.DBView.db2_rep == 'customers' ? 'db2RepSelected' : "db2Rep").attachBrowserEvent("click", fnSelRep);
  var t2 = new sap.m.Text(UtilGen.DBView.createId("db2selSupp"), { customData: [{ key: 'suppliers' }], width: "140px", text: Util.getLangText("JO Reports") }).addStyleClass(UtilGen.DBView.db2_rep == 'suppliers' ? 'db2RepSelected' : "db2Rep").attachBrowserEvent("click", fnSelRep);

  var hb2 = new sap.m.HBox(UtilGen.DBView.createId("db2hbReps0"), {
    wrap: sap.m.FlexWrap.Wrap,
    items: [t1, t2]
  });
  sc.addContent(hb2);

  if (UtilGen.DBView.db2_rep == "customers") {
    createSales();
    var hb = new sap.m.Toolbar(UtilGen.DBView.createId("db2hbReps"), { content: [new sap.m.Text(UtilGen.DBView.createId("db2FromTxt"), { text: 'From' }), fromdt, new sap.m.Text(UtilGen.DBView.createId("db2ToTxt"), { text: 'To' }), todt, new sap.m.Text(UtilGen.DBView.createId("db2RefTxt"), { text: 'Ref.:' }), custname, btLst] })
